        (function(){

            function initAPI(version) {
  var ModAPI = {};
  ModAPI.events = {};
  ModAPI.events.types = ["event"];
  ModAPI.events.listeners = { "event": [] };
  ModAPI.globals = {};
  ModAPI.version = version;

  ModAPI.addEventListener = function AddEventListener(name, callback) {
    if (!callback) {
      throw new Error("Invalid callback!");
    }
    if (ModAPI.events.types.includes(name)) {
      if (!Array.isArray(ModAPI.events.listeners[name])) {
        ModAPI.events.listeners[name] = [];
      }
      ModAPI.events.listeners[name].push(callback);
      console.log("Added new event listener.");
    } else {
      throw new Error("This event does not exist!");
    }
  };

  ModAPI.removeEventListener = function removeEventListener(name, func, slow) {
    if (!func) {
      throw new Error("Invalid callback!");
    }
    if (!Array.isArray(ModAPI.events.listeners[name])) {
      ModAPI.events.listeners[name] = [];
    }
    var targetArr = ModAPI.events.listeners[name];
    if (!slow) {
      if (targetArr.indexOf(func) !== -1) {
        targetArr.splice(targetArr.indexOf(func), 1);
        console.log("Removed event listener.");
      }
    } else {
      var functionString = func.toString();
      targetArr.forEach((f, i) => {
        if (f.toString() === functionString) {
          targetArr.splice(i, 1);
          console.log("Removed event listener.");
        }
      });
    }
  };

  ModAPI.events.newEvent = function newEvent(name) {
    ModAPI.events.types.push(name);
  };

  ModAPI.events.callEvent = function callEvent(name, data) {
    if (
      !ModAPI.events.types.includes(name) ||
      !Array.isArray(ModAPI.events.listeners[name])
    ) {
      if (!Array.isArray(ModAPI.events.listeners[name])) {
        if (ModAPI.events.types.includes(name)) {
          ModAPI.events.listeners.event.forEach((func) => {
            func({ event: name, data: data });
          });
          return;
        }
        return;
      }
      console.error(
        "The ModAPI has been called with an invalid event name: " + name
      );
      console.error("Please report this bug to the repo.");
      return;
    }
    ModAPI.events.listeners[name].forEach((func) => {
      func(data);
    });
    ModAPI.events.listeners.event.forEach((func) => {
      func({ event: name, data: data });
    });

    ModAPI.globals._initUpdate();
  };

  ModAPI.updateComponent = function updateComponent(component) {
    if (
      typeof component !== "string" ||
      ModAPI[component] === null ||
      ModAPI[component] === undefined
    ) {
      return;
    }
    if (!ModAPI.globals || !ModAPI.globals.onGlobalsUpdate) {
      return;
    }
    if (!ModAPI.globals.toUpdate) {
      ModAPI.globals.toUpdate = [];
    }
    if (ModAPI.globals.toUpdate.indexOf(component) === -1) {
      ModAPI.globals.toUpdate.push(component);
    }
  };

  ModAPI.require = function require(component) {
    if (typeof component !== "string") {
      return;
    }
    if (!ModAPI.globals || !ModAPI.globals.onRequire) {
      return;
    }
    ModAPI.globals.onRequire(component);
  };

  ModAPI.globals._initUpdate = function _initUpdate() {
    if (!ModAPI.globals.toUpdate) {
      ModAPI.globals.toUpdate = [];
    }
    ModAPI.globals.toUpdate.forEach((id) => {
      ModAPI.globals.onGlobalsUpdate(id);
    });
    ModAPI.globals.toUpdate = [];
  };
  

  window.ModAPI = ModAPI;
}

  window.ModLoader = function ModLoader(ModsArr) {
    if (!window.eaglerMLoaderMainRun) {
      var searchParams = new URLSearchParams(location.search);
      searchParams.getAll("Mod").forEach((ModToAdd) => {
        console.log(
          "EaglerML: Adding Mod to loadlist from search params: " + ModToAdd
        );
        ModsArr.push(ModToAdd);
      });
      if (
        !!eaglercraftXOpts &&
        !!eaglercraftXOpts.Mods &&
        Array.isArray(eaglercraftXOpts.Mods)
      ) {

          console.log(
            "EaglerML: Adding Mod to loadlist from eaglercraftXOpts: " +
              ModToAdd
          );
          ModsArr.push(ModToAdd);
        });
      }
      window.eaglerMLoaderMainRun = true;
    }
    if (window.noLoadMods === true) {
      ModsArr.splice(0, ModsArr.length);
    }
    function checkModsLoaded(totalLoaded, identifier) {
      console.log(
        "EaglerML: Checking if Mods are finished :: " +
          totalLoaded +
          "/" +
          ModsArr.length
      );
      if (totalLoaded >= ModsArr.length) {
        clearInterval(identifier);
        window.ModGracePeriod = false;
        if (
          window.eaglerMLoaderMainRun &&
          ModAPI &&
          ModAPI.events &&
          ModAPI.events.callEvent
        ) {
          ModAPI.events.callEvent("load", {});
        }
        console.log(
          "EaglerML: Checking if Mods are finished :: All Mods loaded! Grace period off."
        );
      }
    }
    function methodB(currentMod) {
      try {
        console.log("EaglerML: Loading " + currentMod + " via method B.");
        var script = document.createElement("script");
        script.src = currentMod;
        script.setAttribute("data-Mod", currentMod);
        script.setAttribute("data-isMod", true);
        script.onerror = () => {
          console.log(
            "EaglerML: Failed to load " + currentMod + " via method B!"
          );
          script.remove();
          totalLoaded++;
        };
        script.onload = () => {
          console.log(
            "EaglerML: Successfully loaded " + currentMod + " via method B."
          );
          totalLoaded++;
        };
        document.body.appendChild(script);
      } catch (error) {
        console.log(
          "EaglerML: Oh no! The Mod " + currentMod + " failed to load!"
        );
        totalLoaded++;
      }
    }
    window.ModGracePeriod = true;
    var totalLoaded = 0;
    var loaderCheckInterval = null;
    ModsArr.forEach((c) => {
      let currentMod = c;
      console.log("EaglerML: Starting " + currentMod);
      try {
        var req = new XMLHttpRequest();
        req.open("GET", currentMod);
        req.onload = function xhrLoadHandler() {
          console.log("EaglerML: Loading " + currentMod + " via method A.");
          var script = document.createElement("script");
          try {
            script.src =
              "data:text/javascript," + encodeURIComponent(req.responseText);
          } catch (error) {
            methodB(currentMod);
            return;
          }
          script.setAttribute("data-Mod", currentMod);
          script.setAttribute("data-isMod", true);
          script.onerror = () => {
            console.log(
              "EaglerML: Failed to load " + currentMod + " via method A!"
            );
            script.remove();
            totalLoaded++;
          };
          script.onload = () => {
            console.log(
              "EaglerML: Successfully loaded " + currentMod + " via method A."
            );
            totalLoaded++;
          };
          document.body.appendChild(script);
        };
        req.onerror = function xhrErrorHandler() {
          methodB(currentMod);
        };
        req.send();
      } catch (error) {
        methodB(currentMod);
      }
    });
    loaderCheckInterval = setInterval(() => {
      checkModsLoaded(totalLoaded, loaderCheckInterval);
    }, 500);
    console.log(
      "EaglerML: Starting to load " + ModsArr.length + " Mods..."
    );
    window.returntotalloadedmods = function returntotalloadedmods(){
      return totalLoaded
    }
  };
}

      try {

    /* ============================================================
       HUD data bridge — reads the REAL game state that the native
       Java module (TreasureHud.java) publishes each tick to
       localStorage["_eaglercraft_1.12.tchud"]. Everything shown by
       Coordinates / Direction / Ping / Armor / Potions / Held Item
       comes from here, so nothing is ever fabricated: if the Java
       side hasn't reported a value (or reports active=0), the HUD
       shows a placeholder instead of a made-up number.
       ============================================================ */
    var HUD_KEY = '_eaglercraft_1.12.tchud';
    var hud = { active: false };
    function readHudBridge() {
      try {
        var raw = localStorage.getItem(HUD_KEY);
        if (!raw) { hud = { active: false }; return; }
        var txt = b64decodeStr(raw);
        var o = {};
        txt.split('\n').forEach(function (ln) { var i = ln.indexOf('='); if (i > 0) o[ln.substring(0, i)] = ln.substring(i + 1); });
        hud = o; hud.active = (o.active === '1');
      } catch (e) { hud = { active: false }; }
    }
    function compass(yaw) {
      // Minecraft yaw: 0=south(+Z), 90=west(-X), 180=north(-Z), 270=east(+X)
      var dirs8 = ['S','SW','W','NW','N','NE','E','SE'];
      var idx = Math.round(yaw / 45) % 8; if (idx < 0) idx += 8;
      var card = dirs8[idx];
      var axis = { S:'+Z', W:'-X', N:'-Z', E:'+X', SW:'', NW:'', NE:'', SE:'' }[card] || '';
      return { card: card, axis: axis };
    }
    function fmtPrec(v) {
      var n = parseFloat(v); if (isNaN(n)) return '—';
      return (String(state.coordPrecision) === '0') ? String(Math.round(n)) : n.toFixed(1);
    }
    function updateDataHud() {
      if (state.coords) {
        if (hud.active) coordsEl.textContent = 'XYZ  ' + fmtPrec(hud.x) + '  ' + fmtPrec(hud.y) + '  ' + fmtPrec(hud.z);
        else coordsEl.textContent = 'XYZ  — — —';
      }
      if (state.direction) {
        if (hud.active && hud.yaw != null) {
          var c = compass(parseFloat(hud.yaw));
          dirEl.textContent = (state.dirMode === 'detailed' && c.axis) ? (c.card + ' (' + c.axis + ')') : c.card;
        } else dirEl.textContent = '—';
      }
      if (state.ping) {
        var p = hud.active ? parseInt(hud.ping, 10) : -1;
        pingEl.textContent = (p != null && p >= 0) ? (p + ' ms') : 'ping n/a';
      }
      if (state.armorStatus) renderArmor();
      if (state.potionEffects) renderPotions();
      if (state.heldItem) renderHeld();
      if (state.perfMonitor) renderPerfMon();
      if (state.sprintStatus) renderSprint();
      if (state.sneakStatus) renderSneak();
    }
    function barColor(frac) { return frac < 0.25 ? 'low' : ''; }
    function renderArmor() {
      var names = ['Helmet','Chest','Legs','Boots'];
      if (!hud.active || !hud.armor) { armorEl.innerHTML = '<div class="tc-armor-slot"><i>—</i></div>'; return; }
      var slots = hud.armor.split('~');
      var html = '';
      for (var i = 0; i < 4; i++) {
        var s = slots[i] || '-';
        if (s === '-') { html += '<div class="tc-armor-slot"><b>' + names[i] + '</b><i style="color:var(--dim)">—</i></div>'; continue; }
        var parts = s.split('|'); var frac = parseFloat(parts[1]);
        var dur = (frac >= 0) ? Math.round(frac * 100) + '%' : '✓';
        html += '<div class="tc-armor-slot"><b>' + names[i] + '</b><i class="' + (frac >= 0 ? barColor(frac) : '') + '">' + dur + '</i></div>';
      }
      armorEl.innerHTML = html;
    }
    function renderPotions() {
      if (!hud.active || !hud.potions) { potionsEl.innerHTML = ''; return; }
      var rows = hud.potions.split('~').filter(Boolean);
      if (!rows.length) { potionsEl.innerHTML = ''; return; }
      potionsEl.innerHTML = rows.map(function (r) {
        var p = r.split('|'); var name = p[0] || '?'; var amp = parseInt(p[1], 10) || 0; var secs = parseInt(p[2], 10) || 0;
        var lvl = amp > 0 ? ' ' + (amp + 1) : '';
        var mm = Math.floor(secs / 60), ss = secs % 60; var time = mm + ':' + (ss < 10 ? '0' : '') + ss;
        return '<div class="tc-potion-row"><span class="tc-potion-dot"></span><span>' + name + lvl + '</span><span style="color:var(--dim)">' + time + '</span></div>';
      }).join('');

    function loop(now) {
      frames++;

    /* ============================================================
       Keybind Manager — one capture system for every Treasure
       Client hotkey. Buttons carry data-bindkey="menu|sprintToggle

       for conflicts, and stores <name>KeyCode/<name>KeyLabel in
       state (the Mod Menu key stays in menuKeyCode/menuKeyLabel for
       backwards-compat). In-game the hotkeys flip the matching mod.
       ============================================================ */
    var BINDS = {
      menu:        { code: 'menuKeyCode',        label: 'menuKeyLabel' },
      zoom:        { code: 'zoomKeyCode',        label: 'zoomKeyLabel' },
      sprintToggle:{ code: 'sprintToggleKeyCode',label: 'sprintToggleKeyLabel' },

    function labelFor(code) {
      if (!code) return 'Unbound';
      var map = { KeyA:'A',KeyB:'B',KeyC:'C',KeyD:'D',KeyE:'E',KeyF:'F',KeyG:'G',KeyH:'H',KeyI:'I',KeyJ:'J',KeyK:'K',KeyL:'L',KeyM:'M',KeyN:'N',KeyO:'O',KeyP:'P',KeyQ:'Q',KeyR:'R',KeyS:'S',KeyT:'T',KeyU:'U',KeyV:'V',KeyW:'W',KeyX:'X',KeyY:'Y',KeyZ:'Z', ShiftRight:'Right Shift', ShiftLeft:'Left Shift', ControlLeft:'Left Ctrl', ControlRight:'Right Ctrl', AltLeft:'Left Alt', AltRight:'Right Alt', Backslash:'\\', Backquote:'`', Minus:'-', Equal:'=', Tab:'Tab', BracketLeft:'[', BracketRight:']', Semicolon:';', Quote:"'", Comma:',', Period:'.', Slash:'/' };
      if (map[code]) return map[code];
      if (/^F([1-9]|1[0-9]|2[0-4])$/.test(code)) return code;
      if (/^Key([A-Z])$/.test(code)) return code.slice(3);
      if (/^Digit(\d)$/.test(code)) return code.slice(5);
      return code;
    }
    function isSafeMenuKey(code) {
      // The Mod Menu hotkey is captured in the capture phase and swallowed with
      // stopImmediatePropagation before the game sees it, so letters/numbers are
      // fine — you just can't bind it to a couple of keys that would trap you in
      // the menu (Escape closes it) or a modifier that fires constantly.
      if (['Escape'].indexOf(code) !== -1) return false;
      return true;
    }
    function bindConflict(name, code) {
      if (!code) return null;
      var hit = null;
      Object.keys(BINDS).forEach(function (n) { if (n !== name && state[BINDS[n].code] === code) hit = n; });
      return hit;
    }
    function refreshBindButtons() {
      $$('.tc-keybind[data-bindkey]').forEach(function (btn) {
        var name = btn.dataset.bindkey; var b = BINDS[name]; if (!b) return;
        btn.textContent = state[b.label] || labelFor(state[b.code]);
      });
    }

    $$('.tc-keybind[data-bindkey]').forEach(function (btn) {

        $$('.tc-keybind.listening').forEach(function (o) { o.classList.remove('listening'); refreshBindButtons(); });

      });

    function setBind(name, code) {
      var b = BINDS[name]; if (!b) return;
      // The Mod Menu key is required — refuse to clear it (only allow a real key).
      if (name === 'menu' && !code) return;
      state[b.code] = code; state[b.label] = labelFor(code);
      if (name === 'menu') { state.menuKeyCode = code; state.menuKeyLabel = labelFor(code); }
      save(); refreshBindButtons();
    }

      // surface a floating toast so rebinding from the Interface panel (or the

    // GUARD: when the user is typing in one of the overlay's own text inputs
    // (search box, relay code fields, etc.), keep the keystrokes inside that
    // input and stop the game engine from capturing them as movement/hotkeys.
    // Without this the engine swallows the letters and you can't type. Runs in
    // capture phase before everything else. We DON'T stop the bind-capture flow.
    function tcIsTypingTarget(e) {
      var t = e.target;
      if (!t) return false;
      var tag = t.tagName;
      return (tag === 'INPUT' || tag === 'TEXTAREA' || t.isContentEditable) && root.contains(t);
    }
    ['keydown', 'keyup', 'keypress'].forEach(function (evt) {
      window.addEventListener(evt, function (e) {
        // don't intercept while rebinding a key (that flow wants the raw event)
        if (bindListening) return;
        if (tcIsTypingTarget(e)) {
          // The key belongs to one of our text fields. We must NOT stopPropagation
          // in this window capture-phase listener — that would stop the event
          // before it descends to the input itself (the classic "can't type"
          // bug). Instead the input carries its own bubble-phase handlers that
          // stopImmediatePropagation to keep the key away from the game engine
          // once the input has processed it. Here we simply bow out.
          return;
        }
      }, true);
    });

    window.addEventListener('keydown', function (e) {
      if (bindListening) {
        e.preventDefault(); e.stopPropagation();
        var name = bindListening; bindListening = null;
        $$('.tc-keybind.listening').forEach(function (o) { o.classList.remove('listening'); });
        if (e.code === 'Escape') { refreshBindButtons(); return; }
        if (name === 'menu' && !isSafeMenuKey(e.code)) {
          showBindWarn('That key clashes with movement/typing — pick another for the Mod Menu.');
          refreshBindButtons(); return;
        }
        var conflict = bindConflict(name, e.code);
        if (conflict) { showBindWarn('Already bound to “' + conflict + '”. Clear it first.'); refreshBindButtons(); return; }
        setBind(name, e.code);
        return;
      }
      // If the user is typing in one of our text fields, the key belongs to the
      // field — don't treat it as a hotkey (so e.g. a search containing the menu
      // key doesn't close the menu, and letters aren't eaten).
      if (tcIsTypingTarget(e)) return;
      // Mod Menu open/close
      if (state.menuKeyCode && e.code === state.menuKeyCode) { e.preventDefault(); e.stopImmediatePropagation(); toggleGui(); return; }
      // Toggle Sprint / Toggle Sneak are driven by the player's REAL sprint/sneak
      // keys inside the Java module (so a disabled mod's key does nothing, and the
      // latch is handled there) — no separate overlay hotkeys needed here.
      if (e.code === 'Escape' && gui.classList.contains('open')) { e.preventDefault(); closeGui(); }
    }, true);
